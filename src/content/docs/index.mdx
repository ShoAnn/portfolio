export function getYearsSince(pastDateString) {
    const pastDate = new Date(pastDateString);
    const today = new Date();
    let years = today.getFullYear() - pastDate.getFullYear();
    const monthDiff = today.getMonth() - pastDate.getMonth();
    if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < pastDate.getDate())) {
        years--;
    }
    return years;
}
